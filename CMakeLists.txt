cmake_minimum_required(VERSION 3.24)
project(ScheduledUnlock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(CoreTests tests/core_tests.cpp)
  target_include_directories(CoreTests PRIVATE src)
  add_test(NAME request_gates COMMAND CoreTests)
endif()
if(NOT WIN32)
  message(STATUS "Only portable request tests can be built on this platform.")
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This version requires a 64-bit Windows target.")
endif()
enable_language(RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ScheduledUnlockCommon STATIC src/common.cpp)
target_include_directories(ScheduledUnlockCommon PUBLIC src)
target_compile_definitions(ScheduledUnlockCommon PUBLIC UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(ScheduledUnlockCommon PUBLIC advapi32 bcrypt crypt32 wtsapi32 ole32 oleaut32)
add_library(ScheduledUnlockProvider SHARED src/provider.cpp src/provider.def)
target_link_libraries(ScheduledUnlockProvider PRIVATE ScheduledUnlockCommon secur32 uuid)
set_target_properties(ScheduledUnlockProvider PROPERTIES PREFIX "")
add_executable(ScheduledUnlock WIN32 src/app.cpp src/tasks.cpp src/app.rc)
target_link_libraries(ScheduledUnlock PRIVATE ScheduledUnlockCommon taskschd uuid comctl32 shell32 gdi32 user32)
set_target_properties(ScheduledUnlock ScheduledUnlockProvider PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  foreach(target ScheduledUnlockCommon ScheduledUnlockProvider ScheduledUnlock)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /guard:cf)
  endforeach()
  target_link_options(ScheduledUnlock PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /GUARD:CF)
  target_link_options(ScheduledUnlockProvider PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF)
else()
  foreach(target ScheduledUnlockCommon ScheduledUnlockProvider ScheduledUnlock)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
  target_link_options(ScheduledUnlock PRIVATE -municode -static -Wl,--dynamicbase,--nxcompat)
  target_link_options(ScheduledUnlockProvider PRIVATE -static -Wl,--dynamicbase,--nxcompat)
endif()
